    <div style="display: flex; gap: 10px; justify-content: center; align-items: center; flex-wrap: wrap;">
      <input 
        id="email-input"
        type="email"

        style="padding: 12px 20px; width: 280px; border-radius: 6px; border: none; font-size: 14px;"
      >
      <button 
        onclick="captureEmail()"
        style="background: white; color: #667eea; padding: 12px 30px; border: none; border-radius: 6px; font-weight: bold; cursor: pointer; font-size: 14px; transition: transform 0.2s;"

<script>
function captureEmail() {
  const email = document.getElementById('email-input').value;
  
  // Basic email validation
  if (!email || !email.includes('@')) {
    alert('Please enter a valid email address');
    return;
  }
  
  // Save to localStorage (temporary - we'll upgrade this)
  const emails = JSON.parse(localStorage.getItem('chainwatcher_emails') || '[]');
  
  if (!emails.includes(email)) {
    emails.push({
      email: email,
      timestamp: new Date().toISOString(),
      source: document.referrer || 'direct'
    });
    localStorage.setItem('chainwatcher_emails', JSON.stringify(emails));
    
    // Also send to your backend (add this endpoint later)
    fetch('/api/capture-email', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ 
        email: email,
        timestamp: new Date().toISOString(),
        referrer: document.referrer
      })
    }).catch(err => console.log('Backend not ready yet, saved locally'));
  }
  
  // Show success message
  document.getElementById('email-banner').innerHTML = `
    <div style="padding: 20px;">
      <p style="margin: 0; font-size: 20px;">✅ You're in!</p>
      <p style="margin: 10px 0 0 0; font-size: 14px; opacity: 0.9;">
        We'll email you at <strong>${email}</strong> when we detect threats.
      </p>
    </div>
  `;
  
  // Track the conversion
  console.log('Email captured:', email);
  
  // Hide banner after 3 seconds
  setTimeout(() => {
    document.getElementById('email-banner').style.display = 'none';
  }, 3000);
}
</script>
