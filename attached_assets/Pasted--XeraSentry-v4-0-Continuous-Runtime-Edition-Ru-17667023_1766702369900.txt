class ThreatIndicator(Enum):
    GAS_DEVIATION = "gas_deviation"
    VALUE_DEVIATION = "value_deviation"
    RAPID_TRANSACTIONS = "rapid_transactions"
    UNUSUAL_TIME = "unusual_time"
    NEW_ADDRESS = "new_address"
    FAILED_TRANSACTION = "failed_transaction"
    HONEYPOT_CALL = "honeypot_call"
    KNOWN_ATTACKER = "known_attacker"
    BURNER_WALLET = "burner_wallet"
    HIGH_VALUE = "high_value"

@dataclass
class ThreatScore:
    total_score: float
    confidence: str
    reasoning: List[str]
    should_alert: bool
    indicators: Dict[str, float]

@dataclass
class TransactionContext:
    tx_hash: str
    sender: str
    gas_used: int
    value_eth: float
    timestamp: datetime
    function_selector: str
    status: int
    sender_nonce: int
    sender_balance: float
    is_contract: bool

class SmartDetectionEngine:
    def __init__(self):
        self.known_attackers = {
            '0x1da5821544e25c636c1417ba96ade4cf6d2f9b5a',
            '0x3f5ce5fbfe3e9af3971dd833d26ba9b5c936f0be',
        }

        self.mev_bots = {
            '0x000000000035b5e5ad9019092c665357240f594e',
        }

        self.address_history = defaultdict(lambda: {
            'tx_count': 0,
            'first_seen': None,
            'total_value': 0.0
        })

        self.recent_txs = deque(maxlen=1000)

        self.thresholds = {
            'gas_deviation_pct': 10.0,
            'value_deviation_pct': 20.0,
            'min_value_for_alert': 0.5,
            'rapid_tx_count': 5,
            'rapid_tx_window': 60,
            'new_address_tx_limit': 3,
            'burner_balance_limit': 0.1
        }

        self.weights = {
            ThreatIndicator.GAS_DEVIATION: 15,
            ThreatIndicator.VALUE_DEVIATION: 15,
            ThreatIndicator.RAPID_TRANSACTIONS: 25,
            ThreatIndicator.UNUSUAL_TIME: 10,
            ThreatIndicator.NEW_ADDRESS: 15,
            ThreatIndicator.FAILED_TRANSACTION: 30,
            ThreatIndicator.HONEYPOT_CALL: 100,
            ThreatIndicator.KNOWN_ATTACKER: 100,
            ThreatIndicator.BURNER_WALLET: 1.5,
            ThreatIndicator.HIGH_VALUE: 1.3,
        }

    def analyze_transaction(self, ctx: TransactionContext, gas_baseline: float, value_baseline: float) -> ThreatScore:
        if ctx.sender.lower() in self.mev_bots:
            return ThreatScore(0.0, "CLEAN", ["Known MEV bot"], False, {})

        indicators = {}
        reasoning = []
        base_score = 0.0
        multipliers = 1.0

        # High confidence checks
        if ctx.sender.lower() in self.known_attackers:
            indicators['known_attacker'] = 100
            reasoning.append("🚨 KNOWN ATTACKER")
            base_score += 100

        honeypot_selectors = ['0xdeadbeef', '0xbaadface']
        if ctx.function_selector in honeypot_selectors:
            indicators['honeypot_call'] = 100
            reasoning.append("🚨 Honeypot trap triggered")
            base_score += 100

        # Medium confidence
        recent = [tx for tx in self.recent_txs
                 if tx['sender'] == ctx.sender and
                 (ctx.timestamp - tx['timestamp']).total_seconds() < self.thresholds['rapid_tx_window']]

        if len(recent) >= self.thresholds['rapid_tx_count']:
            indicators['rapid_transactions'] = 25
            reasoning.append(f"🤖 Rapid transactions: {len(recent)}+")
            base_score += 25

        if ctx.status == 0:
            indicators['failed_transaction'] = 30
            reasoning.append("❌ Transaction failed")
            base_score += 30

        history = self.address_history[ctx.sender]
        if history['tx_count'] < self.thresholds['new_address_tx_limit'] and ctx.value_eth > self.thresholds['min_value_for_alert']:
            indicators['new_address'] = 15
            reasoning.append(f"🆕 New address with {ctx.value_eth:.4f} ETH")
            base_score += 15

        if 3 <= ctx.timestamp.hour <= 6:
            indicators['unusual_time'] = 10
            reasoning.append(f"🌙 Unusual hour ({ctx.timestamp.hour}:00 UTC)")
            base_score += 10

        # Low confidence
        gas_deviation = abs((ctx.gas_used - gas_baseline) / gas_baseline * 100) if gas_baseline > 0 else 0
        if gas_deviation > self.thresholds['gas_deviation_pct']:
            indicators['gas_deviation'] = 15
            reasoning.append(f"📊 Gas {gas_deviation:.1f}% above baseline")
            base_score += 15

        value_deviation = abs((ctx.value_eth - value_baseline) / value_baseline * 100) if value_baseline > 0 else 0
        if value_deviation > self.thresholds['value_deviation_pct'] and ctx.value_eth > self.thresholds['min_value_for_alert']:
            indicators['value_deviation'] = 15
            reasoning.append(f"💰 Value {value_deviation:.1f}% above baseline")
            base_score += 15

        # Multipliers
        if ctx.sender_nonce < 5 and ctx.sender_balance < self.thresholds['burner_balance_limit']:
            indicators['burner_wallet'] = 1.5
            reasoning.append(f"🔥 Burner wallet")
            multipliers *= 1.5

        if ctx.value_eth > 10.0:
            indicators['high_value'] = 1.3
            reasoning.append(f"💎 High value: {ctx.value_eth:.2f} ETH")
            multipliers *= 1.3

        if ctx.is_contract:
            indicators['contract_interaction'] = 1.2
            reasoning.append("📝 Sender is contract")
            multipliers *= 1.2

        total_score = base_score * multipliers

        # Classify
        if 'known_attacker' in indicators or 'honeypot_call' in indicators:
            confidence = "CRITICAL"
            should_alert = True
        elif total_score >= 96:
            confidence = "CRITICAL"
            should_alert = True
        elif total_score >= 76:
            confidence = "HIGH"
            should_alert = True
        elif total_score >= 51:
            confidence = "MEDIUM"
            should_alert = True
        else:
            confidence = "LOW"
            should_alert = False

        # Update history
        if history['first_seen'] is None:
            history['first_seen'] = ctx.timestamp
        history['tx_count'] += 1
        history['total_value'] += ctx.value_eth

        self.recent_txs.append({
            'sender': ctx.sender,
            'timestamp': ctx.timestamp,
            'function_selector': ctx.function_selector
        })

        return ThreatScore(total_score, confidence, reasoning[:5], should_alert, indicators)

# === MAIN MONITOR ===

class XeraSentry:
    def __init__(self, config: Config):
        self.config = config
        self.running = False

        # Initialize Web3
        logger.info(f"🔌 Connecting to {config.node_url}")
        self.w3 = Web3(Web3.HTTPProvider(config.node_url, request_kwargs={'timeout': 30}))

        if not self.w3.is_connected():
            raise ConnectionError("Failed to connect to Ethereum node")

        logger.info(f"✅ Connected to Ethereum (Chain ID: {self.w3.eth.chain_id})")

        self.addr = Web3.to_checksum_address(config.contract_address)
        logger.info(f"🎯 Monitoring: {config.contract_name}")

        # Initialize components
        self.detection_engine = SmartDetectionEngine()

        self.history = defaultdict(lambda: {
            'gas': deque(maxlen=config.max_history_size),
            'value': deque(maxlen=config.max_history_size)
        })
        self.baselines = {}
        self.local_blacklist = set()
        self.alert_cache = {}

        # Statistics
        self.stats = {
            'start_time': datetime.now(),
            'blocks_processed': 0,
            'transactions_processed': 0,
            'alerts_triggered': 0,
            'alerts_filtered': 0,
            'errors': 0
        }

        # Database
        self._init_db()
        logger.info("🛡️  XeraSentry v4.0 initialized")

    def _init_db(self):
        db_name = 'xerasentry_continuous.db'
        self.conn = sqlite3.connect(db_name, check_same_thread=False)

        self.conn.execute('''CREATE TABLE IF NOT EXISTS alerts
            (id INTEGER PRIMARY KEY AUTOINCREMENT,
             timestamp TEXT NOT NULL,
             tx_hash TEXT NOT NULL,
             contract TEXT NOT NULL,
             threat_type TEXT NOT NULL,
             threat_level TEXT NOT NULL,
             threat_score REAL NOT NULL,
             sender TEXT NOT NULL,
             details TEXT,
             reasoning TEXT,
             UNIQUE(tx_hash, threat_type))''')

        self.conn.execute('''CREATE TABLE IF NOT EXISTS reputation
            (address TEXT PRIMARY KEY,
             alert_count INTEGER DEFAULT 0,
             status TEXT DEFAULT 'CLEAN',
             first_seen TEXT,
             last_alert TEXT)''')

        self.conn.execute('''CREATE TABLE IF NOT EXISTS filtered_alerts
            (id INTEGER PRIMARY KEY AUTOINCREMENT,
             timestamp TEXT,
             tx_hash TEXT,
             sender TEXT,
             score REAL,
             reason TEXT)''')

        self.conn.commit()

    def calculate_mad_threshold(self, data: deque) -> float:
        if not data or len(data) < 15:
            return float('inf')

        try:
            arr = np.array(list(data))
            median = np.median(arr)
            mad = np.median(np.abs(arr - median))

            if mad == 0:
                return median * 1.5

            return median + (6.0 * mad * 1.4826)
        except Exception as e:
            logger.error(f"MAD calculation error: {e}")
            return float('inf')

    def process_transaction(self, tx: Dict, receipt: Dict, silent: bool = False):
        try:
            tx_hash = tx['hash'].hex()
            sender = Web3.to_checksum_address(tx['from'])

            if sender in self.config.whitelist:
                return

            func_selector = (tx['input'][:10] if len(tx.get('input', '')) >= 10 else '0x').lower()
            gas_used = receipt['gasUsed']
            value = float(self.w3.from_wei(tx['value'], 'ether'))

            if not silent and func_selector in self.baselines:
                limits = self.baselines[func_selector]

                try:
                    sender_nonce = self.w3.eth.get_transaction_count(sender)
                    sender_balance = float(self.w3.from_wei(self.w3.eth.get_balance(sender), 'ether'))
                    is_contract = len(self.w3.eth.get_code(sender)) > 0
                except:
                    sender_nonce = 0
                    sender_balance = 0.0
                    is_contract = False

                ctx = TransactionContext(
                    tx_hash=tx_hash,
                    sender=sender,
                    gas_used=gas_used,
                    value_eth=value,
                    timestamp=datetime.now(),
                    function_selector=func_selector,
                    status=receipt.get('status', 1),
                    sender_nonce=sender_nonce,
                    sender_balance=sender_balance,
                    is_contract=is_contract
                )

                threat_score = self.detection_engine.analyze_transaction(
                    ctx,
                    gas_baseline=limits['gas_limit'],
                    value_baseline=limits['val_limit']
                )

                if threat_score.should_alert:
                    self.trigger_alert(tx_hash, sender, threat_score)
                elif threat_score.total_score > 0:
                    self.stats['alerts_filtered'] += 1
                    self.conn.execute(
                        "INSERT INTO filtered_alerts VALUES (NULL, ?, ?, ?, ?, ?)",
                        (datetime.now().isoformat(), tx_hash, sender, threat_score.total_score,
                         " | ".join(threat_score.reasoning[:2]))
                    )
                    self.conn.commit()

            # Update baselines
            self.history[func_selector]['gas'].append(gas_used)
            self.history[func_selector]['value'].append(value)

            self.baselines[func_selector] = {
                'gas_limit': self.calculate_mad_threshold(self.history[func_selector]['gas']),
                'val_limit': self.calculate_mad_threshold(self.history[func_selector]['value'])
            }

            self.stats['transactions_processed'] += 1

        except Exception as e:
            self.stats['errors'] += 1
            logger.error(f"Transaction processing error: {e}")

    def trigger_alert(self, tx_hash: str, sender: str, threat_score: ThreatScore):
        cache_key = f"{sender}:threat"
        last_alert_time = self.alert_cache.get(cache_key, datetime.min)

        if (datetime.now() - last_alert_time).seconds < self.config.alert_cooldown:
            return

        self.alert_cache[cache_key] = datetime.now()

        confidence_emoji = {
            "CRITICAL": "🔴",
            "HIGH": "🟠",
            "MEDIUM": "🟡",
            "LOW": "🟢"
        }

        emoji = confidence_emoji.get(threat_score.confidence, "⚪")

        msg = f"""🛡️  XERASENTRY ALERT
Contract: {self.config.contract_name}
Level: {emoji} {threat_score.confidence}
Score: {threat_score.total_score:.0f}/100
From: {sender[:10]}...{sender[-8:]}

Threat Indicators:
{chr(10).join(f'  {r}' for r in threat_score.reasoning)}

Tx: {tx_hash[:16]}..."""

        logger.warning(msg)

        try:
            self.conn.execute(
                "INSERT OR IGNORE INTO alerts VALUES (NULL, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                (datetime.now().isoformat(), tx_hash, self.addr, "MULTI_FACTOR_THREAT",
                 threat_score.confidence, threat_score.total_score, sender,
                 f"Score: {threat_score.total_score:.0f}",
                 " | ".join(threat_score.reasoning))
            )
            self.conn.commit()
            self.stats['alerts_triggered'] += 1
        except Exception as e:
            logger.error(f"Failed to save alert: {e}")

        if self.config.telegram_bot_token:
            try:
                requests.post(
                    f"https://api.telegram.org/bot{self.config.telegram_bot_token}/sendMessage",
                    json={'chat_id': self.config.telegram_chat_id, 'text': msg},
                    timeout=5
                )
            except Exception as e:
                logger.error(f"Telegram notification failed: {e}")

    def train(self):
        lookback = self.config.training_lookback
        current_block = self.w3.eth.block_number

        logger.info(f"🏋️  Training on last {lookback} blocks...")

        processed = 0
        for block_num in range(current_block - lookback, current_block + 1):
            if not self.running:
                break

            try:
                block = self.w3.eth.get_block(block_num, full_transactions=True)
                for tx in block['transactions']:
                    if tx.get('to') == self.addr:
                        receipt = self.w3.eth.get_transaction_receipt(tx['hash'])
                        self.process_transaction(tx, receipt, silent=True)
                        processed += 1

                if block_num % 10 == 0:
                    logger.info(f"  Training progress: {block_num}/{current_block}")

            except Exception as e:
                if "not found" not in str(e).lower():
                    logger.error(f"Block {block_num} error: {e}")

        logger.info(f"✅ Training complete: {processed} txs, {len(self.baselines)} functions")

    def print_status(self):
        uptime = datetime.now() - self.stats['start_time']

        total_potential = self.stats['alerts_triggered'] + self.stats['alerts_filtered']
        reduction = 0 if total_potential == 0 else (self.stats['alerts_filtered'] / total_potential * 100)

        print(f"\n{'='*60}")
        print(f"📊 XERASENTRY v4.0 STATUS")
        print(f"{'='*60}")
        print(f"⏱️  Uptime: {uptime}")
        print(f"📦 Blocks: {self.stats['blocks_processed']}")
        print(f"🔄 Transactions: {self.stats['transactions_processed']}")
        print(f"🚨 Alerts Triggered: {self.stats['alerts_triggered']}")
        print(f"🔇 Alerts Filtered: {self.stats['alerts_filtered']} ({reduction:.1f}% reduction)")
        print(f"❌ Errors: {self.stats['errors']}")
        print(f"🚫 Blacklisted: {len(self.local_blacklist)}")
        print(f"{'='*60}\n")

    def run(self):
        """Run monitoring continuously until Ctrl+C"""
        logger.info("🚀 Starting XeraSentry - Press Ctrl+C to stop")
        self.running = True

        self.train()

        last_block = self.w3.eth.block_number
        last_status = time.time()

        logger.info(f"📡 Live monitoring active (block {last_block})")
        logger.info(f"⏰ Status updates every {self.config.status_interval} seconds")

        try:
            while self.running:
                try:
                    current_block = self.w3.eth.block_number

                    if current_block > last_block:
                        for block_num in range(last_block + 1, current_block + 1):
                            try:
                                block = self.w3.eth.get_block(block_num, full_transactions=True)

                                for tx in block['transactions']:
                                    if tx.get('to') == self.addr:
                                        receipt = self.w3.eth.get_transaction_receipt(tx['hash'])
                                        self.process_transaction(tx, receipt)

                                self.stats['blocks_processed'] += 1

                            except Exception as e:
                                logger.error(f"Block {block_num} error: {e}")

                        last_block = current_block

                    # Print status periodically
                    if time.time() - last_status >= self.config.status_interval:
                        self.print_status()
                        last_status = time.time()

                    time.sleep(self.config.poll_interval)

                except Exception as e:
                    logger.error(f"Monitoring error: {e}")
                    time.sleep(30)

        except KeyboardInterrupt:
            logger.info("⏸️  Stopped by user (Ctrl+C)")
        finally:
            self.stop()

    def stop(self):
        self.running = False
        self.print_status()
        logger.info("🛑 XeraSentry stopped")
        self.conn.close()
